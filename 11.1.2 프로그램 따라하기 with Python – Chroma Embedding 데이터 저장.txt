  . 질문 임베딩을 기준으로 유사한 문장 검색 
  . 검색 문장/유사도 점수 Return 
4) 주의사항

[ 2. 프로그램 소스 ]

'''
■	프로그램 기본 환경 : Phyton 3.10.10/ Chroma DB 0.6.3
■	사용 모델 : SentenceTransformer version 3.0.1
■	환경 설치 관련 명령어( Pycharm Console 기준 )
     . pip install chromadb 
	 . pip install sentence-transformers
'''

#임포트 및 설정
import chromadb
from sentence_transformers import SentenceTransformer
import warnings
warnings.filterwarnings("ignore", category=FutureWarning, module="transformers.tokenization_utils_base")


#샘플 데이터 로드
sentences = [
        'A stylish 3 bedroom townhouse close to the river with a private balcony.',
        'A serene 4 bedroom cabin in the mountains with a cozy fireplace.',
        'A sleek studio apartment in a vibrant neighborhood with a shared gym.',
        'A charming 2 bedroom cottage in the countryside with scenic views.',
        'A bright 2 bedroom flat near the university with a spacious living area.',
        'A modern 1 bedroom loft in the heart of the city with a rooftop terrace.'
    ]

# 임베딩 생성
model = SentenceTransformer('sentence-transformers/all-MiniLM-L6-v2')
embeddings = model.encode(sentences)

# Chroma DB 클라이언트 초기화
client = chromadb.PersistentClient()

# Collection 생성
collection_name = "Apt103"
collection = client.get_or_create_collection(collection_name)

# 항목 추가 , 항목 embedding 등록  to Chroma DB collection
for idx, description in enumerate(sentences):
    collection.add(
        ids=[str(idx)],
        embeddings=[embeddings[idx].tolist()],  # Chroma DB requires lists for embeddings
        metadatas=[{"description": description}]
   )

# Query 생성: Searching for a property similar to "luxury beach house with a pool"
query = "bright house near a university"
query_embedding = model.encode([query])

# 유사도검색 in Chroma DB
results = collection.query(
    query_embeddings=query_embedding.tolist(),
    n_results=3,  # Number of similar results to return
    include=['metadatas', 'distances']  # Include both metadata and distances
)

# 결과 출력
for idx,(metadata, distance) in enumerate(zip(results['metadatas'][0], results['distances'][0])):
    print(f"Result {idx+1}: {metadata['description']}(Distance: {distance})")
