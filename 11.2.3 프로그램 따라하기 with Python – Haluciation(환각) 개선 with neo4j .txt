    with driver.session() as session:

from openai import OpenAI
import os
from sentence_transformers import SentenceTransformer
from neo4j import GraphDatabase
from numpy import dot
from numpy.linalg import norm

# 1. OpenAI API 설정 (API 키를 환경 변수에서 가져옴)
os.environ["OPENAI_API_KEY"] = "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"
client = OpenAI()

# 2. 임베딩 모델 로드
model = SentenceTransformer('all-MiniLM-L6-v2')  # BERT 기반 모델 예시

# 3. Neo4j 연결 설정
uri = "bolt://localhost:7687"

# 5. 코사인 유사도 계산 함수
def cosine_similarity(vec1, vec2):
    return dot(vec1, vec2) / (norm(vec1) * norm(vec2))

# 6. 유사한 텍스트를 Neo4j에서 검색하는 함수
def find_similar_texts(driver, query_embedding, top_n=3):
    with driver.session() as session:
        # Neo4j에서 모든 텍스트와 임베딩을 가져오기

        result = session.run("MATCH (n:Mytexts) RETURN n.content AS content, n.embedding AS embedding")

        #result = session.run("MATCH(m: Mountain{name: '북한산'})-[:RELATED_TO]->(n:Mytexts) RETURN n.content AS content, n.embedding AS embedding")

        # 유사도를 계산하여 저장할 리스트
        similarities = []

        # 각 텍스트의 임베딩과 쿼리 임베딩 간의 코사인 유사도 계산
        for record in result:
            content = record["content"]
            embedding = record["embedding"]
            similarity = cosine_similarity(query_embedding, embedding)
            similarities.append((content, similarity))

        # 유사도 순으로 상위 n개의 결과 반환
        similarities = sorted(similarities, key=lambda x: x[1], reverse=True)
        return similarities[:top_n]

# 7. 유사한 텍스트 기반으로 OpenAI GPT 모델을 사용해 응답 생성 함수

def generate_response(similarity_results):

    # 유사도 검색 결과를 프롬프트로 포맷팅
    formatted_results = "\n".join([f"Result {i + 1}: {result}" for i, result in enumerate(similarity_results)])

    print(formatted_results )
    # ChatGPT API 호출을 위한 프롬프트 생성
    # Create the prompt for the ChatGPT model


    prompt = f"""
     You are an AI assistant. Based on the following similarity search results, provide a helpful response to the user in Korean :
     Similarity Search Results:
     {formatted_results}
     Answer (Please respond in Korean):
     Example :
     안녕하세요. 답변드립니다.
     xxx는 xxx 이고, xxx이며, '''입니다.
     추가적인 질문이 있으시면 언제든지 말씀해 주세요."
     """

    # OpenAI API 호출
    response = client.chat.completions.create(
        model="gpt-4o-mini",  # 사용할 GPT 모델 (예: 'gpt-4' 또는 'gpt-3.5-turbo')
        messages=[
            {"role": "system", "content": "You are a helpful assistant."},
            {"role": "user", "content": prompt}
        ],
        max_tokens=500,  # 출력의 최대 길이
        temperature= 1.0  # 창의성을 조정하는 파라미터 (0.0 ~ 1.0)
    )

    # 응답에서 텍스트 추출
    answer = response.choices[0].message.content
    return answer

# 8. Neo4j에 연결하여 유사한 텍스트 검색
driver = GraphDatabase.driver(uri, auth=(user, password))
similar_texts = find_similar_texts(driver, query_embedding)

# 9. 검색된 유사한 텍스트 기반으로 답변 생성
response = generate_response(similar_texts)
print(f"Generated Response: {response}")

# 10. 드라이버 닫기
driver.close()
