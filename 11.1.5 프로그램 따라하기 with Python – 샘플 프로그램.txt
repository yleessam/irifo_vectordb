[ 2. 프로그램 소스 ]

# 필요한 라이브러리 임포트
from langchain.text_splitter import RecursiveCharacterTextSplitter
from sentence_transformers import SentenceTransformer
import chromadb
import warnings
import os
from openai import OpenAI

# 경고 메시지 억제
warnings.filterwarnings("ignore", category=FutureWarning, module="transformers.tokenization_utils_base")

# 상수 정의
CHUNK_SIZE = 256  # 텍스트 청크의 크기
CHUNK_OVERLAP = 20  # 청크 간 겹치는 문자 수
EMBEDDING_MODEL_NAME = 'sentence-transformers/all-MiniLM-L6-v2'  # 임베딩 생성 모델
OPENAI_API_KEY = "OpenAI API 키를 입력"  # OpenAI API 키 (실제 키로 대체 필요)
CHROMA_COLLECTION_NAME = "Apt012" # ChromaDB 컬렉션 이름

# OpenAI API 초기화
os.environ["OPENAI_API_KEY"] = OPENAI_API_KEY
openai_client = OpenAI()

# 유틸리티 함수 및 클래스 정의
class TextChunkProcessor:
    """텍스트를 처리하여 관리 가능한 청크로 분할하는 클래스 (LangChain의 텍스트 분할기 사용)."""
    def __init__(self, chunk_size=CHUNK_SIZE, chunk_overlap=CHUNK_OVERLAP):
        self.text_splitter = RecursiveCharacterTextSplitter(
            chunk_size=chunk_size,
            chunk_overlap=chunk_overlap
        )

    def split_text(self, text):
        """텍스트를 청크로 분할."""
        documents = self.text_splitter.create_documents([text])
        return [doc.page_content for doc in documents]


class TextEmbedder:
    """SentenceTransformer 모델을 사용하여 텍스트 임베딩을 생성하는 클래스."""
    def __init__(self, model_name=EMBEDDING_MODEL_NAME):
        self.model = SentenceTransformer(model_name)

    def generate_embeddings(self, texts):
        """텍스트 리스트에 대한 임베딩 생성."""
        return self.model.encode(texts)


class ChromaDBHandler:
    """ChromaDB를 활용해 데이터 저장 및 검색을 처리하는 클래스."""
    def __init__(self, collection_name):
        self.client = chromadb.PersistentClient()
        self.collection = self.client.get_or_create_collection(collection_name)

    def add_to_collection(self, ids, embeddings, metadatas):
        """ChromaDB 컬렉션에 데이터 추가."""
        self.collection.add(ids=ids, embeddings=embeddings, metadatas=metadatas)

    def query_collection(self, query_embedding, n_results=2):
        """ChromaDB 컬렉션에서 유사도 검색."""
        return self.collection.query(
            query_embeddings=query_embedding.tolist(),
            n_results=n_results,
            include=['metadatas', 'distances']
        )


def generate_chatgpt_response(query, similarity_results):
    """유사도 검색 결과를 기반으로 OpenAI GPT를 사용해 응답 생성."""
    # 유사도 검색 결과를 포맷팅
    formatted_results = "\n".join([f"Result {i + 1}: {result}" for i, result in enumerate(similarity_results)])

    # ChatGPT에 보낼 프롬프트 생성
    prompt = f"""
    당신은 AI 어시스턴트입니다. 다음의 유사도 검색 결과를 기반으로 사용자가 이해할 수 있도록 한국어로 답변하세요:
    유사도 검색 결과:
    {formatted_results}

    답변 (한국어로 작성):
    질문 "{query}" 에 대한 답변은 아래와 같습니다.
    """

    # OpenAI API 호출
    response = openai_client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[
            {"role": "system", "content": "You are a helpful assistant."},
            {"role": "user", "content": prompt}
        ],
        max_tokens=500,
        temperature=1.0
    )

    # GPT 응답 반환
    return response.choices[0].message.content


# 메인 워크플로우
if __name__ == "__main__":
    # 입력 텍스트 정의
    text = """
    The housing business outlook for the metropolitan area and non-metropolitan areas is showing contrasting trends. Housing business operators judged that the housing business in the metropolitan area would improve compared to the previous month,
    and predicted that the non-metropolitan area would worsen further.
    According to the Housing Industry Research Institute's housing business outlook index for housing business operators this month, the national index fell 4.6 points (p) from the previous month to 81.6. The metropolitan area is expected to rise 1.3p to 107.4, while the non-metropolitan area is expected to show an overall downward trend, falling 5.9p to 76.0. If the housing business outlook index exceeds the baseline of 100, it means that the proportion of companies that expect the housing business to improve is high, and if it falls below 100, it means the opposite.

    Among the metropolitan areas, Gyeonggi showed the largest increase nationwide, rising 8.0p (102.5→110.5), and Incheon maintained the baseline (100) for three consecutive months without change, while
    Seoul is expected to fall 4.0p to 111.9. Amid the ongoing shortage of supply compared to demand, apartment prices in Seoul and its neighboring areas are still on the rise, and
    expectations for an interest rate cut in the US are interpreted as having a positive effect on business sentiment.
    In the case of non-metropolitan areas, it is expected to fall 5.9p to 76.0, but all metropolitan areas fell, dropping an average of 8.9p (87.7→78.8), and provincial areas fell an average of 3.6p (77.6→74.0). It is highly likely that the additional burden on the market will be due to the strengthening of lending such as the Stress DSR Stage 2 regulation and the increase in the housing loan interest rate, and the slow recovery of housing prices in non-metropolitan areas, which is interpreted as having a negative impact on the psychology of business owners.

    All metropolitan areas fell, with Daejeon showing the largest decline at 17.7p (100.0→82.3). It was followed by Daegu at 17.6p (95.8→78.2), Gwangju at 11.1p (66.6→55.5),
    Busan at 4.9p (80.9→76.0), Ulsan at 1.9p (89.4→87.5), and Sejong at 0.4p (93.7→93.3). All provincial regions declined except Gyeongnam 6.7p (66.6→73.3), Gyeongbuk 5.2p (73.3→78.5), and Chungnam 4.6p (80.0→84..6). Gangwon 20.0p (100.0→80.0), Chungbuk 12.7p (72.7→60.0), Jeonbuk 6.4p (83.3→76.9), Jeju 4.2p (76.4→72.2), and Jeonnam 2.1p (68.7→66.6). Daejeon’s index appears to have declined due to the prolonged stagnation of the local housing market along with the regional economic downturn.
    """

    # 1단계: 텍스트 분할
    processor = TextChunkProcessor()
    sentences = processor.split_text(text)

    # 2단계: 임베딩 생성
    embedder = TextEmbedder()
    embeddings = embedder.generate_embeddings(sentences)

    # 3단계: ChromaDB에 데이터 저장
    chroma_handler = ChromaDBHandler(CHROMA_COLLECTION_NAME)
    chroma_handler.add_to_collection(
        ids=[str(i) for i in range(len(sentences))],  # 각 텍스트에 고유 ID 부여
        embeddings=[emb.tolist() for emb in embeddings],  # 임베딩을 리스트로 변환하여 저장
        metadatas=[{"description": sentence} for sentence in sentences]  # 메타데이터로 텍스트 저장
    )

    # 4단계: 데이터 쿼리
    query = "What was the rate of decline in Daejeon's index?"  # 검색 쿼리 정의
    query_embedding = embedder.generate_embeddings([query])  # 쿼리 임베딩 생성
    search_results = chroma_handler.query_collection(query_embedding)  # 유사도 검색 수행

    # 5단계: GPT 응답 생성
    metadata_results = search_results['metadatas'][0]  # 검색된 메타데이터 추출
    formatted_results = [result['description'] for result in metadata_results]  # 결과 포맷팅
    answer = generate_chatgpt_response(query, formatted_results)  # GPT를 사용해 응답 생성

    # 6단계: 결과 출력
    print("\n생성된 응답:\n", answer)
